  <script>
    // Open Modal and Increment View Count
    function openLightbox(imgSrc, itemId) {
      const modal = document.getElementById('lightbox');
      const modalImg = document.getElementById('lightbox-img');
      const modalDl = document.getElementById('lightbox-dl');
      
      modalImg.src = imgSrc;
      modalDl.href = imgSrc;
      modal.classList.add('active');

      // Increment View Counter
      const card = document.getElementById(itemId);
      const viewCounter = card.querySelector('.view-count');
      let currentViews = parseInt(viewCounter.innerText);
      viewCounter.innerText = currentViews + 1;
    }

    function closeLightbox() {
      document.getElementById('lightbox').classList.remove('active');
    }

    // Toggle Reaction (Like)
    function toggleLike(itemId) {
      const card = document.getElementById(itemId);
      const likeBtn = card.querySelector('.like-btn');
      const likeCounter = card.querySelector('.like-count');
      let currentLikes = parseInt(likeCounter.innerText);

      if (likeBtn.classList.contains('liked')) {
        likeBtn.classList.remove('liked');
        likeCounter.innerText = currentLikes - 1;
      } else {
        likeBtn.classList.add('liked');
        likeCounter.innerText = currentLikes + 1;
      }
    }

    // Category Filter
    function filterGallery(category) {
      const buttons = document.querySelectorAll('.filter-btn');
      buttons.forEach(btn => btn.classList.remove('active'));
      event.target.classList.add('active');

      const items = document.querySelectorAll('.gallery-item');
      items.forEach(item => {
        if (category === 'all' || item.classList.contains(category)) {
          item.style.display = 'flex';
        } else {
          item.style.display = 'none';
        }
      });
    }
  </script>
